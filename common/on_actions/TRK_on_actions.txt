on_actions = {
    on_subject_free = {
            effect = {
                if = {
                    limit = { 
                        original_tag = TMS
                    }
                
				drop_cosmetic_tag = yes
				
            }
        }
    }

    on_ruling_party_change = {
		effect = {
			if = {
				limit = {
                    original_tag = TUR
					NOT = {	has_government = communism }
				}
				drop_cosmetic_tag = yes
			}
		}	
	}

    on_ruling_party_change = {
		effect = {
            if = {
                limit = { 
                    original_tag = TUR
                    ruling_party = communism
                }
                set_cosmetic_tag = TUR_communism_clr
            }
		}
	}

 }
